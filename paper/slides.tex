\begin{frame}{The Phenomenon: Loss of Plasticity (LoP)}
\vspace{-2mm}

\begin{columns}[onlytextwidth,T]
\begin{column}{0.46\textwidth}
\textbf{The Continual Learning Problem}

\smallskip
In continual streams, networks retain old knowledge but \textbf{lose the ability to fit future data}.

\vspace{2mm}
\textbf{The Empirical Symptoms}

\smallskip
Across architectures, as accuracy drops:
\begin{itemize}
    \item \textcolor{frozen}{\textbf{Dead units}} rise (e.g., ResNet)
    \item \textcolor{cloned}{\textbf{Duplicate features}} emerge (e.g., ViT)
    \item \textbf{Effective rank collapses}
\end{itemize}

\vspace{2mm}
\begin{block}{}
\centering
\emph{Why does GD fail to recover once this state appears?}
\end{block}
\end{column}

\begin{frame}{The Hypothesis: LoP as a Topological Trap}

\begin{columns}[onlytextwidth,c]
\begin{column}{0.45\textwidth}
Prior work catalogs \emph{symptoms}. We mathematically prove the \textbf{mechanism of persistence}.

\vspace{4mm}
A manifold $\mathcal{M}\subset\Theta$ induces LoP if the gradient is permanently tangent to it:
\[
\nabla_\theta \ell(\theta;x,y)\in \T \mathcal{M} \quad \forall(x,y)
\]

\vspace{2mm}
\textbf{Key Insight:}
The lock is enforced by the \emph{network equations}, not the data. Once inside, discrete GD/SGD updates remain trapped in $\mathcal{M}$ forever, completely independent of the data distribution.
\end{column}

\begin{frame}{Theorem 1: The Exact Mechanics of Entrapment}
\vspace{-2mm}

\textbf{Degeneracy in the computation} $\Rightarrow$ \textbf{degeneracy in the parameter update.}

\vspace{1mm}
\begin{columns}[onlytextwidth,T]
\begin{column}{0.48\textwidth}
    \textbf{\textcolor{frozen}{1. Frozen Manifold ($\mathcal{M}_F$)}}

    \smallskip
    If $f'(z_v)=0$ persistently, incoming gradients vanish for every sample.

    \smallskip
    \emph{\textcolor{frozen}{A frozen unit is permanently silent to backprop.}}

    \vspace{3mm}
    \textbf{\textcolor{cloned}{2. Cloning Manifold ($\mathcal{M}_C$)}}

    \smallskip
    If block pairs satisfy equal row/col sums, $h_u\!=\!h_{u'}$ and $\delta_v\!=\!\delta_{v'}$ strictly.

    \smallskip
    \emph{\textcolor{cloned}{Cloned directions are indistinguishable. Gradients locked forever.}}
\end{column}

\begin{column}{0.48\textwidth}
    \centering
    \textbf{Empirical Proof: SGD \& Adam are Trapped}
    \vspace{1mm}

    \includegraphics[width=0.75\linewidth]{images/mlp_cloning_r2_plot.pdf}

    \vspace{1mm}
    {\small\textbf{Even Adam is trapped.} SGD and Adam achieve $R^2\!=\!1.0$ for 400k steps. Escape requires explicit symmetry breaking.}
\end{column}
\end{columns}

\end{frame}

% SLIDE 5: Theorem 2 (Rank-Plasticity Tension)
% =========================================================================
\begin{frame}{The Cause: The Rank-Plasticity Tension (Theorem 2)}

\textbf{Why do models fall in?} Training balances \textbf{compression} (for generalization) and \textbf{nonlinear creation} (for diversity). \emph{The very mechanisms that help a model generalize on a static task actively drive it into LoP manifolds.}

\vspace{2mm}
\begin{columns}[onlytextwidth, c]
\begin{column}{0.48\textwidth}
    \begin{block}{Rank-Gain Theorem}
    \centering
    Feature Creation = \\
    \textcolor{frozen}{\textbf{Strength ($\gamma$)}} $\times$ \textcolor{cloned}{\textbf{Opportunity ($\Psi$)}}
    \end{block}

    \vspace{4mm}
    \textbf{The Edge States of Feature Learning:}
    \begin{itemize}
        \item Maximize strength $\gamma \uparrow$ causes $\mathcal{M}_F$ (\textcolor{frozen}{\textbf{Frozen Trap}}).
        \vspace{2mm}
        \item Compress opportunity $\Psi \downarrow$ causes $\mathcal{M}_C$ (\textcolor{cloned}{\textbf{Cloned Trap}}).
    \end{itemize}
\end{column}

\begin{column}{0.50\textwidth}
    \centering
    \textbf{Empirical Proof of the Tension}\\
    \vspace{1mm}
    \includegraphics[width=0.98\linewidth, trim=0 380bp 0 30bp, clip]{images/universal_lop_plot.pdf}\\
    \vspace{1mm}
    {\scriptsize Regimes maximizing decorrelation strength drive units toward a high probability of freezing (proven across 6 activations).}
\end{column}
\end{columns}

\end{frame}
